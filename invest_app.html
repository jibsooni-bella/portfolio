export default { async fetch(request) { try { const url = new URL(request.url); if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders() }); } const ticker = (url.searchParams.get("ticker") || "").trim().toUpperCase(); const type = (url.searchParams.get("type") || "").trim().toLowerCase(); const range = (url.searchParams.get("range") || "1W").trim().toUpperCase(); if (!ticker) { return jsonResponse({ error: "ticker is required" }, 400); } const normalizedTicker = normalizeYahooTicker(ticker); if (type === "history") { const historyData = await fetchYahooHistory(normalizedTicker, range); return jsonResponse({ ticker, normalizedTicker, range, prices: historyData }); } const price = await fetchYahooCurrentPriceFromChart(normalizedTicker); return jsonResponse({ ticker, normalizedTicker, price }); } catch (err) { return jsonResponse( { error: err.message || "worker error" }, 500 ); } } }; function corsHeaders() { return { "content-type": "application/json; charset=UTF-8", "access-control-allow-origin": "*", "access-control-allow-methods": "GET, OPTIONS", "access-control-allow-headers": "*" }; } function jsonResponse(data, status = 200) { return new Response(JSON.stringify(data), { status, headers: corsHeaders() }); } function normalizeYahooTicker(rawTicker) { const t = String(rawTicker || "").trim().toUpperCase(); if (t === "KRX-GOLD") return "132030.KS"; if (/^\d{6}$/.test(t)) return `${t}.KS`; return t; } function getRangeConfig(range) { switch (range) { case "1W": return { interval: "1h", period1: daysAgoToUnix(7), period2: nowUnix() }; case "1M": return { interval: "1d", period1: daysAgoToUnix(30), period2: nowUnix() }; case "3M": return { interval: "1d", period1: daysAgoToUnix(90), period2: nowUnix() }; case "6M": return { interval: "1d", period1: daysAgoToUnix(180), period2: nowUnix() }; case "12M": return { interval: "1d", period1: daysAgoToUnix(370), period2: nowUnix() }; case "24M": return { interval: "1wk", period1: daysAgoToUnix(740), period2: nowUnix() }; case "36M": return { interval: "1wk", period1: daysAgoToUnix(1100), period2: nowUnix() }; case "60M": return { interval: "1wk", period1: daysAgoToUnix(1850), period2: nowUnix() }; case "MAX": return { interval: "1mo", period1: 0, period2: nowUnix() }; default: return { interval: "1h", period1: daysAgoToUnix(7), period2: nowUnix() }; } } function nowUnix() { return Math.floor(Date.now() / 1000); } function daysAgoToUnix(days) { return Math.floor((Date.now() - days * 24 * 60 * 60 * 1000) / 1000); } async function fetchYahooCurrentPriceFromChart(ticker) { const period2 = nowUnix(); const period1 = period2 - (10 * 24 * 60 * 60); const chartUrl = `https://query1.finance.yahoo.com/v8/finance/chart/${encodeURIComponent(ticker)}` + `?period1=${period1}&period2=${period2}&interval=1d&includePrePost=false&events=div%2Csplits`; const res = await fetch(chartUrl, { headers: { "user-agent": "Mozilla/5.0" } }); if (!res.ok) { throw new Error(`현재가 조회 실패 (${res.status})`); } const data = await res.json(); const result = data?.chart?.result?.[0]; const closes = result?.indicators?.quote?.[0]?.close || []; const validCloses = closes.filter(v => typeof v === "number" && Number.isFinite(v)); if (!validCloses.length) { throw new Error("현재가 데이터를 찾지 못했어."); } return validCloses[validCloses.length - 1]; } async function fetchYahooHistory(ticker, range) { const config = getRangeConfig(range); const chartUrl = `https://query1.finance.yahoo.com/v8/finance/chart/${encodeURIComponent(ticker)}` + `?period1=${config.period1}&period2=${config.period2}&interval=${config.interval}&includePrePost=false&events=div%2Csplits`; const res = await fetch(chartUrl, { headers: { "user-agent": "Mozilla/5.0" } }); if (!res.ok) { throw new Error(`히스토리 조회 실패 (${res.status})`); } const data = await res.json(); const result = data?.chart?.result?.[0]; const timestamps = result?.timestamp || []; const closes = result?.indicators?.quote?.[0]?.close || []; if (!timestamps.length || !closes.length) { throw new Error("히스토리 데이터가 없어."); } const prices = []; for (let i = 0; i < timestamps.length; i++) { const ts = timestamps[i]; const close = closes[i]; if (typeof close !== "number" || !Number.isFinite(close)) continue; prices.push({ date: unixToDate(ts), close }); } if (!prices.length) { throw new Error("유효한 히스토리 데이터가 없어."); } return prices; } function unixToDate(ts) { const d = new Date(ts * 1000); const y = d.getUTCFullYear(); const m = String(d.getUTCMonth() + 1).padStart(2, "0"); const day = String(d.getUTCDate()).padStart(2, "0"); return `${y}-${m}-${day}`; }
